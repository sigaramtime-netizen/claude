+@route(r"/muhasebe/hesap-durumu", roles=())
+def muhasebe_hesap_durumu(req):
+    """D023-M4 madde 9-11 — Hesap Durumu matrisi: hesap × 12 ay.
+
+    Satır başı açılış, satır sonu kapanış bakiyesi (aktif/pasife göre işaret);
+    hücreler ay borç / ay alacak; alt satır aylık Σborç/Σalacak/net.
+    Parametreler: yıl seçimi + 'kümülatif bakiye göster' (mizan bakiye deseni: borç−alacak).
+    """
+    conn = db.get_conn()
+    sid = db.sirket_id(req)
+    bugun = datetime.date.today()
+    yil = _i(req.q("yil")) or bugun.year
+    kumulatif = req.q("kumulatif") == "1"
+    hesaplar = conn.execute(
+        "SELECT id, kod, ad, tip FROM hesap WHERE aktif=1 AND sirket_id=? ORDER BY kod",
+        (sid,)).fetchall()
+    secili_idler = []
+    for v in (req.query.get("hesap") or []):
+        try:
+            secili_idler.append(int(v))
+        except (TypeError, ValueError):
+            pass
+    if req.q("tumu") == "1" or not secili_idler:
+        secili_idler = [h["id"] for h in hesaplar]
+    secili_set = set(secili_idler)
+    # açılış bakiyeleri (yıl başından önceki tüm hareketler)
+    acilis_map = {}
+    for r in conn.execute(
+            "SELECT k.hesap_id, COALESCE(SUM(k.borc),0) b, COALESCE(SUM(k.alacak),0) a "
+            "FROM yevmiye_kalem k JOIN yevmiye y ON y.id=k.yevmiye_id "
+            "WHERE y.durum='Onaylandı' AND y.sirket_id=? AND y.tarih < ? GROUP BY k.hesap_id",
+            (sid, f"{yil}-01-01")):
+        acilis_map[r["hesap_id"]] = round(r["b"] - r["a"], 2)
+    # yıl içi aylık hareketler
+    ay_map = {}
+    for r in conn.execute(
+            "SELECT k.hesap_id, substr(y.tarih,1,7) ay, COALESCE(SUM(k.borc),0) b, "
+            "COALESCE(SUM(k.alacak),0) a FROM yevmiye_kalem k "
+            "JOIN yevmiye y ON y.id=k.yevmiye_id "
+            "WHERE y.durum='Onaylandı' AND y.sirket_id=? AND y.tarih >= ? AND y.tarih <= ? "
+            "GROUP BY k.hesap_id, substr(y.tarih,1,7)",
+            (sid, f"{yil}-01-01", f"{yil}-12-31")):
+        ay_map[(r["hesap_id"], (r["ay"] or "")[:7])] = (round(r["b"], 2), round(r["a"], 2))
+    rows = []
+    for h in hesaplar:
+        if h["id"] not in secili_set:
+            continue
+        acilis = acilis_map.get(h["id"], 0.0)
+        aylar, tb, ta, kum = [], 0.0, 0.0, acilis
+        for m in range(1, 13):
+            b, a = ay_map.get((h["id"], f"{yil}-{m:02d}"), (0.0, 0.0))
+            kum = round(kum + b - a, 2)
+            aylar.append({"ay": m, "b": b, "a": a,
+                          "kum_g": _bk_goster(kum, h["tip"])})
+            tb = round(tb + b, 2)
+            ta = round(ta + a, 2)
+        kapanis = round(acilis + tb - ta, 2)
+        rows.append({"id": h["id"], "kod": h["kod"], "ad": h["ad"], "tip": h["tip"],
+                     "acilis": acilis, "acilis_g": _bk_goster(acilis, h["tip"]),
+                     "aylar": aylar, "tb": tb, "ta": ta, "kapanis": kapanis,
+                     "kapanis_g": _bk_goster(kapanis, h["tip"])})
+    toplam = {"acilis": 0.0, "tb": 0.0, "ta": 0.0, "kapanis": 0.0, "aylar": []}

+        a = round(sum(r["aylar"][m - 1]["a"] for r in rows), 2)
+        toplam["aylar"].append({"ay": m, "b": b, "a": a, "net": round(b - a, 2)})
+    toplam["acilis"] = round(sum(r["acilis"] for r in rows), 2)
+    toplam["tb"] = round(sum(r["tb"] for r in rows), 2)
+    toplam["ta"] = round(sum(r["ta"] for r in rows), 2)
+    toplam["kapanis"] = round(sum(r["kapanis"] for r in rows), 2)
+    yillar = list(range(bugun.year - 5, bugun.year + 2))
+    conn.close()
+    return render_template("muhasebe/hesap_durumu.html", rows=rows, toplam=toplam,
+                           hesaplar=hesaplar, secili=secili_idler, yil=yil, yillar=yillar,
+                           kumulatif=kumulatif)
