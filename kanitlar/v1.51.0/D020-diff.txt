+def _check_rebuild(conn, tablo, degisiklikler, trigger_sql, indexler):
+    """CHECK değiştiren migrasyonlar için genel tablo rebuild (D019 deseninin geneli).
+
+    degisiklikler: [(eski_check, yeni_check)] — DDL'de birebir replace edilir.
+    Tırnaklı/tırnaksız/IF NOT EXISTS CREATE varyantlarının üçü de desteklenir
+    (D019'da yakalanan çift-replace tuzağına karşı if/elif/else).
+    """
+    row = conn.execute(
+        "SELECT sql FROM sqlite_master WHERE type='table' AND name=?", (tablo,)).fetchone()
+    if not row:
+        return False
+    yeni = row["sql"] or ""
+    for eski, yenis in degisiklikler:
+        yeni = yeni.replace(eski, yenis, 1)
+    if yeni == (row["sql"] or ""):
+        return False
+    if f'CREATE TABLE "{tablo}"' in yeni:
+        yeni = yeni.replace(f'CREATE TABLE "{tablo}"', f"CREATE TABLE {tablo}_yeni", 1)
+    elif f"CREATE TABLE IF NOT EXISTS {tablo}" in yeni:
+        yeni = yeni.replace(f"CREATE TABLE IF NOT EXISTS {tablo}",
+                            f"CREATE TABLE IF NOT EXISTS {tablo}_yeni", 1)
+    else:
+        yeni = yeni.replace(f"CREATE TABLE {tablo}", f"CREATE TABLE {tablo}_yeni", 1)
+    conn.commit()
+    conn.execute("PRAGMA foreign_keys = OFF")
+    try:
+        conn.execute(yeni)
+        conn.execute(f"INSERT INTO {tablo}_yeni SELECT * FROM {tablo}")
+        conn.execute(f"DROP TABLE {tablo}")
+        conn.execute(f"ALTER TABLE {tablo}_yeni RENAME TO {tablo}")
+    finally:
+        conn.commit()
+        conn.execute("PRAGMA foreign_keys = ON")
+    if trigger_sql:
+        conn.executescript(trigger_sql)
+    for ix in indexler:
+        conn.execute(ix)
+    return True
